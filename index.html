        function checkSession() {

            const loginTime = sessionStorage.getItem('admin_login_time');

            if (!loginTime) return true; // Jika belum login, abaikan


            const timeout = 60 * 60 * 1000; // 1 jam dalam milidetik

            const now = Date.now();

            const adminPage = document.getElementById('page-admin-dashboard');


            // Jika halaman admin tidak aktif, jangan logout

            if (!adminPage || adminPage.classList.contains('hidden')) {

                return true;

            }


            if (now - parseInt(loginTime) > timeout) {

                // Hapus SEMUA data session

                sessionStorage.removeItem('admin_login_time');

                sessionStorage.removeItem('admin_nama');

                return false;

            }

            return true;

        }

        function renderTable(elementId, transactions) {

            const tbody = document.getElementById(elementId);

            if (!tbody) return;

            tbody.innerHTML = '';


            if (!transactions || transactions.length === 0) {

                tbody.innerHTML = `<tr><td colspan="4" class="text-center py-6 text-slate-400">Tidak ada transaksi pada periode ini.</td></tr>`;

                return;

            }


            // === LOGIKA GROUPING ID INDUK (.1, .2, .3) ===

            const groupedData = {};

            

            transactions.forEach(trx => {

                // Cari ID induk (misal TRX-0017.1 -> TRX-0017)

                let parentId = trx.id || trx.noBukti || 'TRX-0000';

                if (parentId.includes('.')) {

                    parentId = parentId.split('.')[0]; // Ambil bagian depan titik

                }

                

                // Kelompokkan berdasarkan ID induk

                if (!groupedData[parentId]) {

                    groupedData[parentId] = {

                        items: [],

                        totalPemasukan: 0,

                        totalPengeluaran: 0,

                        tanggal: trx.tanggal,

                        jenis: trx.jenis,

                        linkBukti: trx.linkBukti || '',

                        // === PERBAIKAN: Simpan data untuk membuat keterangan gabungan ===

                        baseKeterangan: trx.keterangan, // Simpan keterangan item pertama

                        bulanList: [] // Array untuk menampung daftar bulan

                    };

                }

                groupedData[parentId].items.push(trx);

                groupedData[parentId].totalPemasukan += trx.pemasukan || 0;

                groupedData[parentId].totalPengeluaran += trx.pengeluaran || 0;


                // === PERBAIKAN: Ekstrak nama bulan dari keterangan setiap item ===

                // Asumsi format keterangan di database: "Iuran Air - 120 - Anto (Mei)"

                const match = trx.keterangan.match(/\(([^)]+)\)$/);

                if (match && match[1]) {

                    // Cegah duplikasi bulan jika ada kesalahan data

                    if (!groupedData[parentId].bulanList.includes(match[1])) {

                        groupedData[parentId].bulanList.push(match[1]);

                    }

                }

            });


            // Render hasil pengelompokan (satu baris per ID induk)

            Object.values(groupedData).forEach(group => {

                let isMasuk = group.jenis?.toLowerCase() === 'pemasukan' || false;

                let iconBg = isMasuk ? 'bg-emerald-50 text-emerald-500' : 'bg-rose-50 text-rose-500';

                    ? `<span class="flex items-center gap-1.5"><span class="w-1.5 h-1.5 rounded-full bg-emerald-500"></span> <span class="text-xs font-semibold text-slate-600">Masuk</span></span>`

                    : `<span class="flex items-center gap-1.5"><span class="w-1.5 h-1.5 rounded-full bg-rose-500"></span> <span class="text-xs font-semibold text-slate-600">Keluar</span></span>`;


                let nominalClass = isMasuk ? 'text-slate-800' : 'text-slate-500';

                

                // === PERBAIKAN LOGIKA DISPLAY KETERANGAN ===

                if (group.bulanList && group.bulanList.length > 0) {

                    // Hapus kurung bulan yang lama (jika ada) agar tidak double

                    let baseString = group.baseKeterangan.replace(/\([^)]+\)$/, '').trim();

                    // Jika baseString kosong setelah dihapus, gunakan string aslinya sebagai fallback

                    if (baseString === '') baseString = group.baseKeterangan;

                    // Gabungkan dengan daftar bulan baru yang sudah diekstrak

                    displayKeterangan = `${baseString} (${group.bulanList.join(', ')})`;

        function renderAdminTable() {

            const bulanEl = document.getElementById('adm-filter-bulan');

            const tahunEl = document.getElementById('adm-filter-tahun');

            const jenisEl = document.getElementById('adm-filter-jenis');

            

            const selectedMonth = bulanEl ? bulanEl.value : "08";

            const selectedYear = tahunEl ? tahunEl.value : "2026";

            const selectedJenis = jenisEl ? jenisEl.value : "all";


            // === TAMBAHKAN 3 BARIS INI UNTUK UPDATE TEKS SALDO AWAL ===

            const monthNames = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];

            if (document.getElementById('adm-ket-awal')) {

                document.getElementById('adm-ket-awal').innerText = `Per 01 ${monthNames[parseInt(selectedMonth)-1]} ${selectedYear}`;

            }

            

            const tbody = document.getElementById('adm-table-body');

            if(!tbody) return;

            tbody.innerHTML = '';


            // Jika data transaksi kosong, reset kartu menjadi 0

            if(!adminTrxData || adminTrxData.length === 0) {

                tbody.innerHTML = `<tr><td colspan="9" class="text-center py-6 text-slate-400">Belum ada data transaksi di Spreadsheet.</td></tr>`;

                // Reset kartu ringkasan

                if(document.getElementById('adm-saldo-awal')) document.getElementById('adm-saldo-awal').innerText = 'Rp 0';

                if(document.getElementById('adm-total-masuk')) document.getElementById('adm-total-masuk').innerText = 'Rp 0';

                if(document.getElementById('adm-total-keluar')) document.getElementById('adm-total-keluar').innerText = 'Rp 0';

                if(document.getElementById('adm-saldo-akhir')) document.getElementById('adm-saldo-akhir').innerText = 'Rp 0';

                return;

            }


            // === LOGIKA GROUPING ID INDUK (.1, .2, .3) UNTUK TABEL ADMIN ===

            const groupedData = {};

            

            adminTrxData.forEach(trx => {

                // 1. Cari ID induk

                let parentId = trx.noBukti || 'TRX-0000';

                if (parentId.includes('.')) parentId = parentId.split('.')[0];


                // 2. Kelompokkan berdasarkan ID induk

                if (!groupedData[parentId]) {

                    groupedData[parentId] = {

                        noBukti: parentId,

                        tanggal: trx.tanggal,

                        jenis: trx.jenis,

                        baseUraian: trx.uraian.replace(/\([^)]+\)$/, '').trim(),

                        bulanList: [],

                        totalPemasukan: 0,

                        totalPengeluaran: 0,

                        items: []

                    };

                }


                const match = trx.uraian.match(/\(([^)]+)\)$/);

                if (match && match[1]) {

                    if (!groupedData[parentId].bulanList.includes(match[1])) {

                        groupedData[parentId].bulanList.push(match[1]);

                    }

                }


                groupedData[parentId].items.push(trx);

                groupedData[parentId].totalPemasukan += trx.pemasukan || 0;

                groupedData[parentId].totalPengeluaran += trx.pengeluaran || 0;

            });


            // Filter berdasarkan jenis (harus dilakukan pada GROUP)

            let filteredGroups = Object.values(groupedData);

            if (selectedJenis !== 'all') {

                filteredGroups = filteredGroups.filter(g => g.jenis.toLowerCase() === selectedJenis.toLowerCase());

            }

            // Filter bulan & tahun secara manual pada tanggal group

            filteredGroups = filteredGroups.filter(g => {

                const parts = g.tanggal.split('/');

                if (parts.length !== 3) return false;

                return (parts[1] === selectedMonth && parts[2] === selectedYear);

            });


            // === HITUNG SALDO BERJALAN & RENDER ===

            let runningSaldo = 0;

            let saldoAwal = 0;

            let totalMasuk = 0, totalKeluar = 0;


            // 1. Hitung Saldo Awal (semua transaksi SEBELUM bulan dan tahun terpilih)

            let tempSaldo = 0;

            adminTrxData.forEach(trx => {

                const parts = String(trx.tanggal).split('/');

                if(parts.length === 3) {

                    let m = parts[1], y = String(parts[2]);

                    let masuk = Number(trx.pemasukan) || 0, keluar = Number(trx.pengeluaran) || 0;

                    if (y < selectedYear || (y === selectedYear && m < selectedMonth)) {

                        tempSaldo += (masuk - keluar);

                    }

                }

            });

            

            // === PERBAIKAN PENTING DI SINI ===

            saldoAwal = tempSaldo; // Masukkan hasil perhitungan ke variabel saldoAwal

            runningSaldo = saldoAwal;


            if(filteredGroups.length === 0) {

                const monthNames = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];

                const monthText = monthNames[parseInt(selectedMonth)-1] + " " + selectedYear;

                tbody.innerHTML = `<tr><td colspan="9" class="text-center py-6 text-slate-400">Tidak ada transaksi pada bulan ${monthText}.</td></tr>`;

                

                // Update Kartu Ringkasan meskipun tidak ada transaksi di bulan ini

                if(document.getElementById('adm-saldo-awal')) document.getElementById('adm-saldo-awal').innerText = formatRp(saldoAwal);

                if(document.getElementById('adm-total-masuk')) document.getElementById('adm-total-masuk').innerText = 'Rp 0';

                if(document.getElementById('adm-total-keluar')) document.getElementById('adm-total-keluar').innerText = 'Rp 0';

                if(document.getElementById('adm-saldo-akhir')) document.getElementById('adm-saldo-akhir').innerText = formatRp(saldoAwal);

                return;

            }


            // 2. Proses Filtered Groups untuk bulan berjalan

            filteredGroups.forEach(group => {

                runningSaldo = runningSaldo + group.totalPemasukan - group.totalPengeluaran;

                totalMasuk += group.totalPemasukan;

                totalKeluar += group.totalPengeluaran;


                let displayUraian = group.baseUraian;

                if (group.bulanList && group.bulanList.length > 0) {

                    displayUraian = `${group.baseUraian} (${group.bulanList.join(', ')})`;

                }


                let isMasuk = group.jenis.toLowerCase() === 'pemasukan';

                let badgeClass = isMasuk ? 'bg-emerald-50 text-emerald-600' : 'bg-rose-50 text-rose-600';

                

                let actionButtons = `

                    <button onclick="editTransaksi('${group.noBukti}')" class="w-7 h-7 rounded-full bg-blue-50 text-blue-600 hover:bg-blue-100 border border-blue-100 flex items-center justify-center transition-all shadow-sm" title="Edit Transaksi">

                        <i class="fa-regular fa-pen-to-square text-[10px]"></i>

                    </button>

                    <button onclick="hapusTransaksi('${group.noBukti}')" class="w-7 h-7 rounded-full bg-rose-50 text-rose-600 hover:bg-rose-100 border border-rose-100 flex items-center justify-center transition-all shadow-sm" title="Hapus Transaksi">

                        <i class="fa-regular fa-trash-can text-[10px]"></i>

                    </button>

                `;


                tbody.innerHTML += `

                    <tr class="hover:bg-slate-50/50 transition-colors">

                        <td class="py-3 px-3 text-slate-600 font-medium">${group.tanggal}</td>

                        <td class="py-3 px-3 text-blue-600 font-bold">${group.noBukti}</td>

                        <td class="py-3 px-3 text-slate-800 font-semibold">${displayUraian}</td>

                        <td class="py-3 px-3"><span class="px-2.5 py-1 rounded-md text-[10px] font-bold uppercase ${badgeClass}">${group.jenis}</span></td>

                        <td class="py-3 px-3 text-right font-bold text-emerald-600">${group.totalPemasukan > 0 ? formatRp(group.totalPemasukan) : '-'}</td>

                        <td class="py-3 px-3 text-right font-bold text-rose-600">${group.totalPengeluaran > 0 ? formatRp(group.totalPengeluaran) : '-'}</td>

                        <td class="py-3 px-3 text-right font-extrabold text-slate-800">${formatRp(runningSaldo)}</td>

                        <td class="py-3 px-3 text-slate-500">${group.bulanList.join(', ') || '-'}</td>

                        <td class="py-3 px-3 text-center">

                            <div class="flex items-center justify-center gap-2">

                                ${actionButtons}

                            </div>

                        </td>

                    </tr>

                `;

            });


            // 3. Update Kartu Ringkasan Di Bagian Paling Bawah

            let saldoAkhir = saldoAwal + totalMasuk - totalKeluar;

            

            if(document.getElementById('adm-saldo-awal')) document.getElementById('adm-saldo-awal').innerText = formatRp(saldoAwal);

            if(document.getElementById('adm-total-masuk')) document.getElementById('adm-total-masuk').innerText = formatRp(totalMasuk);

            if(document.getElementById('adm-total-keluar')) document.getElementById('adm-total-keluar').innerText = formatRp(totalKeluar);

            if(document.getElementById('adm-saldo-akhir')) document.getElementById('adm-saldo-akhir').innerText = formatRp(saldoAkhir);

            if(document.getElementById('adm-foot-masuk')) document.getElementById('adm-foot-masuk').innerText = formatRp(totalMasuk);

        // FUNGSI UNTUK MENGEKSEKUSI PENGHAPUSAN (SUDAH DIPERBAIKI)

        function executeDelete() {

                showNotification("Gagal", "ID Transaksi tidak ditemukan.", "error");

                return;

            }

            

            // 1. Simpan ID ke variabel lokal agar tidak hilang

            const idYangAkanDihapus = pendingDeleteId;


            // 2. Tutup pop-up konfirmasi (ini akan mereset pendingDeleteId jadi null)

            closeConfirmModal(); 


            // 3. Tampilkan status loading di tabel

            const tbody = document.getElementById('adm-table-body');

            if(tbody) tbody.innerHTML = `<tr><td colspan="9" class="text-center py-6 text-slate-400"><i class="fa-solid fa-circle-notch fa-spin text-blue-500 text-2xl mb-2"></i><br>Sedang menghapus data...</td></tr>`;

            google.script.run

                .withSuccessHandler(function(res) {

                    if (res.success) {

                        showNotification("Berhasil", res.message, "success");

                    } else {

                        showNotification("Gagal", res.message, "error");

                    }

                    loadAdminBukuKas(); 

                })

                .withFailureHandler(function(err) {

                    showNotification("Koneksi Error", "Gagal menghapus: " + err.message, "error");

                    loadAdminBukuKas();

                })

                .deleteTransactionByID(idYangAkanDihapus); // <-- Menggunakan variabel baru

        }

        function populateAdminYearDropdown() {

            const yearDropdown = document.getElementById('adm-filter-tahun');

            if (!yearDropdown) return;

            

            // Simpan tahun yang sedang dipilih pengguna (agar tidak reset saat refresh data)

            const currentVal = yearDropdown.value;

            

            // Kosongkan dropdown

            yearDropdown.innerHTML = '';

            

            // Jika data kosong, berikan default tahun 2026

            if (!adminTrxData || adminTrxData.length === 0) {

                const opt = document.createElement('option');

                opt.value = "2026";

                opt.innerText = "2026";

                yearDropdown.appendChild(opt);

                yearDropdown.value = "2026";

                return;

            }


            // 1. Kumpulkan semua tahun unik dari data transaksi

            const years = new Set();

            adminTrxData.forEach(trx => {

                const parts = String(trx.tanggal).split('/');

                if (parts.length === 3) {

                    const y = parts[2];

                    if (y && y.length === 4) { // Pastikan format tahun 4 digit (YYYY)

                        years.add(y);

                    }

                }

            });


            // 2. Urutkan tahun dari yang paling tua ke paling baru

            const sortedYears = Array.from(years).sort((a, b) => a - b);


            if (sortedYears.length === 0) {

                // Fallback jika tidak ada tahun yang valid

                const opt = document.createElement('option');

                opt.value = "2026";

                opt.innerText = "2026";

                yearDropdown.appendChild(opt);

                yearDropdown.value = "2026";

                return;

            }


            // 3. Tambahkan opsi ke dropdown

            sortedYears.forEach(y => {

                const opt = document.createElement('option');

                opt.value = y;

                opt.innerText = y;

                yearDropdown.appendChild(opt);

            });


            // 4. Set pilihan default:

            // - Jika pengguna sebelumnya memilih tahun tertentu dan tahun itu ada, gunakan itu.

            // - Jika tidak, pilih tahun terbaru (paling akhir di array sortedYears).

            if (currentVal && sortedYears.includes(currentVal)) {

                yearDropdown.value = currentVal;

            } else {

                yearDropdown.value = sortedYears[sortedYears.length - 1];

            }

        }
